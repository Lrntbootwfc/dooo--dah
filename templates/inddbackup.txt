    <script>
        let currentUser = null;
        let vFileSystem = []; 
        let currentFileId = null;
        let clipboardAsset = null; 

        document.addEventListener("DOMContentLoaded", () => {
            const rememberedUser = localStorage.getItem("vault_current_user");
            if (rememberedUser) { currentUser = rememberedUser; bootWorkspace(); }
            const savedAvatar = localStorage.getItem("comic_avatar_desc");
            if (savedAvatar) { document.getElementById("avatarConfig").value = savedAvatar; }
        });

        function changeTheme(themeName) { document.body.setAttribute("data-theme", themeName); }

        function toggleSidebar() {
            const sidebar = document.getElementById("sidebarNode");
            const workspace = document.getElementById("workspaceNode");
            sidebar.classList.toggle("collapsed");
            workspace.style.paddingLeft = sidebar.classList.contains("collapsed") ? "40px" : "80px";
        }

        function handleAuth() {
            const u = document.getElementById("authUsername").value.trim();
            if(!u) return alert("Username tracking designation cannot be blank!");
            currentUser = u;
            localStorage.setItem("vault_current_user", u);
            
            if (!localStorage.getItem(`recursive_fs_${currentUser}`)) {
                const initialSeed = [
                    { id: "fold_seed_1", type: "folder", name: "Chai Logs", children: [
                        { id: "file_seed_1", type: "file", name: "First Hack Sprint", content: "Code compiled smoothly on the first attempt.", mood: "💻", created: new Date().toLocaleString(), edited: new Date().toLocaleString(), comic: "" }
                    ] }
                ];
                localStorage.setItem(`recursive_fs_${currentUser}`, JSON.stringify(initialSeed));
            }
            
            const authScreen = document.getElementById("authScreen");
            authScreen.style.opacity = "0";
            setTimeout(() => {
                authScreen.classList.add("hidden");
                bootWorkspace();
            }, 500);
        }

        function bootWorkspace() {
            document.getElementById("appInterface").classList.remove("hidden");
            document.getElementById("hamburgerWrapper").classList.remove("hidden");
            vFileSystem = JSON.parse(localStorage.getItem(`recursive_fs_${currentUser}`));
            renderTreeGrid();
        }

        function logout() { localStorage.removeItem("vault_current_user"); location.reload(); }

        function executeRichFormat(command, value = null) {
            document.execCommand(command, false, value);
            updateMetadataCounters();
        }

        function updateMetadataCounters() {
            const text = document.getElementById("editorEngine").innerText.trim();
            document.getElementById("charCount").innerText = `Characters: ${text.length}`;
            document.getElementById("wordCount").innerText = `Words: ${text === "" ? 0 : text.split(/\s+/).length}`;
        }

        function renderTreeGrid() {
            const container = document.getElementById("treeView");
            container.innerHTML = "";
            container.appendChild(buildRecursiveDOM(vFileSystem));
            localStorage.setItem(`recursive_fs_${currentUser}`, JSON.stringify(vFileSystem));
        }

        function buildRecursiveDOM(nodes) {
            const wrapper = document.createElement("div");
            nodes.forEach(node => {
                const block = document.createElement("div");
                block.className = "node-block";

                const row = document.createElement("div");
                row.className = `node-row ${currentFileId === node.id ? 'active' : ''}`;
                
                const titleSpan = document.createElement("span");
                titleSpan.className = "node-title";
                titleSpan.innerHTML = node.type === "folder" ? `📁 <b>${node.name}</b>` : `📄 ${node.name}`;
                if(node.type === "file") { titleSpan.onclick = () => openDocument(node.id); }
                row.appendChild(titleSpan);

                const dots = document.createElement("span");
                dots.className = "dots-menu-btn";
                dots.innerText = "⋮";
                dots.onclick = (e) => { e.stopPropagation(); closeAllContextMenus(); menuBox.style.display = "block"; };
                row.appendChild(dots);

                const menuBox = document.createElement("div");
                menuBox.className = "context-menu";
                
                if(node.type === "folder") {
                    menuBox.innerHTML = `
                        <div onclick="addChildFile('${node.id}')">+ New File</div>
                        <div onclick="addChildFolder('${node.id}')">+ New Folder</div>
                        <div onclick="renameNode('${node.id}')">Rename</div>
                        <div onclick="pasteNode('${node.id}')">Paste Asset</div>
                        <div onclick="deleteNode('${node.id}')" style="color:#ff4444;">Delete</div>
                    `;
                } else {
                    menuBox.innerHTML = `
                        <div onclick="cutNode('${node.id}')">Cut / Move</div>
                        <div onclick="renameNode('${node.id}')">Rename</div>
                        <div onclick="deleteNode('${node.id}')" style="color:#ff4444;">Delete</div>
                    `;
                }
                row.appendChild(menuBox);
                block.appendChild(row);

                if(node.type === "folder" && node.children && node.children.length > 0) {
                    const subTree = buildRecursiveDOM(node.children);
                    subTree.className = "file-list";
                    block.appendChild(subTree);
                }
                wrapper.appendChild(block);
            });
            return wrapper;
        }

        function closeAllContextMenus() { document.querySelectorAll('.context-menu').forEach(m => m.style.display = "none"); }

        function addRootFolder() {
            const t = prompt("Name your new structural root folder:");
            if (!t) return;
            vFileSystem.push({ id: "fold_" + Date.now(), type: "folder", name: t, children: [] });
            renderTreeGrid();
        }

        function findNodeRecursive(nodes, id) {
            for(let node of nodes) {
                if(node.id === id) return node;
                if(node.type === "folder" && node.children) {
                    let found = findNodeRecursive(node.children, id);
                    if(found) return found;
                }
            }
            return null;
        }

        function removeNodeRecursive(nodes, id) {
            for(let i=0; i<nodes.length; i++) {
                if(nodes[i].id === id) return nodes.splice(i, 1)[0];
                if(nodes[i].type === "folder" && nodes[i].children) {
                    let removed = removeNodeRecursive(nodes[i].children, id);
                    if(removed) return removed;
                }
            }
            return null;
        }

        function addChildFile(parentFolderId) {
            const t = prompt("Enter file label title:");
            if(!t) return;
            const parent = findNodeRecursive(vFileSystem, parentFolderId);
            const stamp = new Date().toLocaleString();
            const newFile = { id: "file_" + Date.now(), type: "file", name: t, content: "Start writing...", mood: "😊", created: stamp, edited: stamp, comic: "" };
            parent.children.push(newFile);
            renderTreeGrid();
            openDocument(newFile.id);
        }

        function addChildFolder(parentFolderId) {
            const t = prompt("Enter subfolder layout label name:");
            if(!t) return;
            const parent = findNodeRecursive(vFileSystem, parentFolderId);
            parent.children.push({ id: "fold_" + Date.now(), type: "folder", name: t, children: [] });
            renderTreeGrid();
        }

        function renameNode(id) {
            const target = findNodeRecursive(vFileSystem, id);
            const n = prompt("Input updated designation string:", target.name);
            if(n) { target.name = n; renderTreeGrid(); if(currentFileId === id) document.getElementById("openTitle").innerText = "📄 " + n; }
        }

        function deleteNode(id) {
            if(confirm("Confirm action: Extract deletion routines on targeted tree node?")) {
                removeNodeRecursive(vFileSystem, id);
                if(currentFileId === id) {
                    currentFileId = null;
                    document.getElementById("openTitle").innerText = "No File Selection Target Active";
                    document.getElementById("editorEngine").innerHTML = "Select an operational folder leaf document element node...";
                    document.getElementById("timestampBox").innerText = "";
                    document.getElementById("comicBoard").classList.add("hidden");
                }
                renderTreeGrid();
            }
        }

        function cutNode(id) {
            clipboardAsset = removeNodeRecursive(vFileSystem, id);
            document.getElementById("clipboardStatus").innerText = `Moving: [${clipboardAsset.name}]`;
            renderTreeGrid();
        }

        function pasteNode(targetFolderId) {
            if(!clipboardAsset) return alert("Clipboard system memory registers empty!");
            const targetParent = findNodeRecursive(vFileSystem, targetFolderId);
            targetParent.children.push(clipboardAsset);
            clipboardAsset = null;
            document.getElementById("clipboardStatus").innerText = "";
            renderTreeGrid();
        }

        function openDocument(id) {
            currentFileId = id;
            const doc = findNodeRecursive(vFileSystem, id);
            if(doc) {
                document.getElementById("openTitle").innerText = "📄 " + doc.name;
                document.getElementById("editorEngine").innerHTML = doc.content;
                document.getElementById("docMood").value = doc.mood;
                document.getElementById("timestampBox").innerText = `Created: ${doc.created}  |  Last Revised: ${doc.edited}`;
                updateMetadataCounters();
                
                if (doc.comic) {
                    document.getElementById("comicBoard").classList.remove("hidden");
                    document.getElementById("comicFrameTarget").src = doc.comic;
                } else {
                    document.getElementById("comicBoard").classList.add("hidden");
                }
            }
            renderTreeGrid();
        }

        function saveActiveDocument() {
            if (!currentFileId) return alert("Select active file endpoint path before calling data sync updates!");
            const doc = findNodeRecursive(vFileSystem, currentFileId);
            if (doc) {
                doc.content = document.getElementById("editorEngine").innerHTML;
                doc.mood = document.getElementById("docMood").value;
                doc.edited = new Date().toLocaleString(); 
                document.getElementById("timestampBox").innerText = `Created: ${doc.created}  |  Last Revised: ${doc.edited}`;
            }
            renderTreeGrid();
            alert("Record details securely committed to workspace memory.");
        }

        function renderAICohesiveComic() {
            if (!currentFileId) return alert("Bring target document node into active focus before requesting art rendering!");
            const rawContent = document.getElementById("editorEngine").innerText;
            const currentMood = document.getElementById("docMood").value;
            const avatarDescription = document.getElementById("avatarConfig").value.trim();
            
            if(!rawContent.trim() || rawContent.startsWith("Select an existing")) {
                return alert("Pen down some narrative logs before requesting art renders!");

            localStorage.setItem("comic_avatar_desc", avatarDescription);
            const frame = document.getElementById("comicBoard");
            const img = document.getElementById("comicFrameTarget");
            frame.classList.remove("hidden");
            
            let characterInjectionString = avatarDescription !== "" ? `Featuring a recurring character styled exactly as a ${avatarDescription}. ` : "";
            const optimizedSystemPrompt = `A sequential 4-panel indie graphic novel page layout grid. High contrast black and white comic ink sketch style, clean crisp dark linework, highly detailed background scenery. ${characterInjectionString}Depicting the following action story arc scene context reflecting a mood signature of ${currentMood}: ${rawContent}. Absolutely no text, no speech bubbles, no typography inside the art cells.`;

        function downloadActiveComicPanel() {
            const imgUrl = document.getElementById("comicFrameTarget").src;

            a.href = imgUrl;

            a.click();

        function getAllFilesRecursive(nodes) {
            let files = [];
            nodes.forEach(node => {
                if(node.type === "file") { files.push(node); }
                else if(node.type === "folder" && node.children) { files = files.concat(getAllFilesRecursive(node.children)); }
            });
            return files;
        }

        function compileMonthlyComicPDF() {
            const targetMonthStr = prompt("Enter compilation target month and year filter (e.g. '6/2026' or 'June'):");
            if(!targetMonthStr) return;

            const allFiles = getAllFilesRecursive(vFileSystem);
            const compilableFiles = allFiles.filter(file => file.created.toLowerCase().includes(targetMonthStr.toLowerCase()) && file.comic && file.comic !== "");

            if (compilableFiles.length === 0) return alert(`No unlocked comic panels found matching the chronological target milestone folder matrix for: ${targetMonthStr}`);

            alert(`Compiling ${compilableFiles.length} sequential journal panels into a digital comic layout... Please wait.`);
            const { jsPDF } = window.jspdf;
            const pdf = new jsPDF('p', 'mm', 'a4');
            
            compilableFiles.forEach((file, index) => {
                if (index > 0) pdf.addPage();
                pdf.setDrawColor(0, 0, 0); pdf.setLineWidth(1.5); pdf.rect(10, 10, 190, 277);
                pdf.setFillColor(0, 0, 0); pdf.rect(10, 10, 190, 15, 'F');
                pdf.setTextColor(255, 255, 255); pdf.setFont("helvetica", "bold"); pdf.setFontSize(14);
                pdf.text(`CHAPTER: ${file.name.toUpperCase()}`, 15, 20);
                
                pdf.setFillColor(240, 240, 240); pdf.rect(10, 267, 190, 20, 'F'); pdf.rect(10, 267, 190, 20, 'D');
                pdf.setTextColor(0, 0, 0); pdf.setFont("helvetica", "normal"); pdf.setFontSize(10);
                pdf.text(`Chronicle Stamp: ${file.created} | Mood: ${file.mood}`, 15, 279);
                
                try { pdf.addImage(file.comic, 'PNG', 15, 35, 180, 180); } 
                catch(err) { pdf.rect(15, 35, 180, 180); pdf.text("Securing cross-origin art vectors... [Asset Bound]", 50, 120); }
            });

            pdf.save(`Chronicle_Comic_Book_${targetMonthStr.replace('/', '_')}.pdf`);
            alert("Your digital comic compilation has been successfully structured and saved!");
        }
